% ============================================================
% C Programming Language
% ============================================================
\lstdefinelanguage{CVSCode}{
  language = C,
  deletekeywords = {for, while, if, else, return, switch, case, break, continue, do, goto},
  % [1] blue: keywords
  morekeywords = [1]{int, char, void, long, short, unsigned, signed, float, double,
    const, static, struct, sizeof, auto, enum, union, typedef, extern, volatile, register},
  % [2] purple: control flow
  morekeywords = [2]{for, while, if, else, return, switch, case, default, break,
    continue, do, goto},
  % [3] teal: types
  morekeywords = [3]{bool, size_t, ssize_t, FILE, int8_t, int16_t, int32_t, int64_t,
    uint8_t, uint16_t, uint32_t, uint64_t},
  % [4] yellow: common functions
  morekeywords = [4]{main, printf, scanf, fprintf, sprintf, snprintf, malloc, calloc,
    realloc, free, memcpy, memset, strlen, strcmp, strcpy, fopen, fclose, puts, getchar},
  morecomment = [l]{//},
}

% ============================================================
% R Programming Language
% ============================================================
\lstdefinelanguage{RVSC}{
  % Treat "." and "_" as identifier letters so data.frame, paste0, NA_integer_ match as one token.
  alsoletter = {._},
  % [1] purple: control flow
  morekeywords = [1]{if, else, for, while, repeat, in, next, break, return},
  % [2] yellow: common functions
  morekeywords = [2]{library, require, source, c, list, data.frame, matrix, vector,
    factor, print, cat, paste, paste0, length, nrow, ncol, sum, mean, seq, rep, apply,
    sapply, lapply},
  % [3] blue: function keyword and constants
  morekeywords = [3]{function, TRUE, FALSE, NULL, NA, Inf, NaN,
    NA_integer_, NA_real_, NA_complex_, NA_character_},
  sensitive = true,
  morecomment = [l]{\#},
  morestring = [b]",
  morestring = [b]',
  moredelim = [s][\color{vsvar}]{`}{`},
}
